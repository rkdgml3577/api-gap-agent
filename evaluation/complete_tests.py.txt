from fastapi.testclient import TestClient
from sample_service.app import create_app

def test_quantity_boundaries():
    with TestClient(create_app()) as client:
        for quantity, expected in [(0, 400), (1, 201), (100, 201), (101, 400)]:
            assert client.post("/orders", headers={"X-User": "alice"}, json={"quantity": quantity}).status_code == expected

def test_shipping_cancel_rejected_and_state_preserved():
    with TestClient(create_app()) as client:
        assert client.post("/orders/2/cancel", headers={"X-User": "alice"}).status_code == 409
        assert client.get("/orders/2", headers={"X-User": "alice"}).json()["status"] == "shipping"
